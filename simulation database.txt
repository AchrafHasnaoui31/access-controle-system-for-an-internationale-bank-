-- 0.6  Supprimer les rôles PostgreSQL (nettoyage complet)
DO $$
DECLARE r RECORD;
BEGIN
    FOR r IN
        SELECT rolname FROM pg_roles
        WHERE rolname LIKE 'banque_%'
    LOOP
        EXECUTE 'DROP ROLE IF EXISTS ' || quote_ident(r.rolname);
    END LOOP;
END;
$$;

-- 3.3  Association rôles <-> permissions
CREATE TABLE role_permissions (
    id_role       INT  NOT NULL REFERENCES roles(id_role)       ON DELETE CASCADE,
    id_permission INT  NOT NULL REFERENCES permissions(id_permission) ON DELETE CASCADE,
    PRIMARY KEY (id_role, id_permission)
);

-- Relation clientèle — conseillers
INSERT INTO role_permissions
  SELECT r.id_role, p.id_permission
  FROM roles r, permissions p
  WHERE r.nom_role IN ('conseiller_particulier','conseiller_entreprise')
    AND p.ressource_type IN ('DONNEES_CLIENT','DOSSIER_KYC','OPERATIONS_COURANTES','CONTRAT_CREDIT')
    AND p.action IN ('LIRE','ECRIRE');

-- Relation clientèle — responsable agence
INSERT INTO role_permissions
  SELECT r.id_role, p.id_permission
  FROM roles r, permissions p
  WHERE r.nom_role = 'responsable_agence'
    AND p.ressource_type IN ('DONNEES_CLIENT','DOSSIER_KYC','OPERATIONS_COURANTES','KPI_COMMERCIAL')
    AND p.action IN ('LIRE','MODIFIER','VALIDER');

-- Relation clientèle — guichetier responsable
INSERT INTO role_permissions
  SELECT r.id_role, p.id_permission
  FROM roles r, permissions p
  WHERE r.nom_role = 'guichetier_responsable'
    AND (
      (p.ressource_type IN ('DONNEES_CLIENT','DOSSIER_KYC','OPERATIONS_COURANTES')
       AND p.action IN ('LIRE','ECRIRE','VALIDER'))
      OR (p.ressource_type = 'CONTRAT_CREDIT'       AND p.action = 'LIRE')
      OR (p.ressource_type = 'KPI_COMMERCIAL'       AND p.action = 'LIRE')
      OR (p.ressource_type = 'JOURNAL_AUDIT'        AND p.action = 'ECRIRE')
      OR  p.ressource_type = 'OPERATIONS_GUICHET'
    )
ON CONFLICT DO NOTHING;

    'Habilitation initiale - equipe ' || e.nom
FROM utilisateurs u
JOIN equipes e ON u.id_equipe = e.id_equipe
ON CONFLICT (id_utilisateur) DO UPDATE
    SET niveau_max = EXCLUDED.niveau_max,
        commentaire = EXCLUDED.commentaire;

FROM   blp_habilitations h

-- 13.1  Droits effectifs d'un utilisateur (RBAC)
CREATE OR REPLACE VIEW v_droits_utilisateur AS
SELECT
    u.matricule,
    u.nom || ' ' || u.prenom   AS utilisateur,
    e.nom                       AS equipe,
    r.nom_role                  AS role,
    p.ressource_type,
    p.action,
    p.niveau_min                AS niveau_requis,
    h.niveau_max                AS niveau_habilite
FROM utilisateurs u
JOIN utilisateur_roles ur ON u.id_utilisateur = ur.id_utilisateur AND ur.actif = TRUE
JOIN roles         r      ON ur.id_role = r.id_role
JOIN equipes       e      ON r.id_equipe = e.id_equipe
JOIN role_permissions rp  ON r.id_role = rp.id_role
JOIN permissions   p      ON rp.id_permission = p.id_permission
JOIN blp_habilitations h  ON u.id_utilisateur = h.id_utilisateur
WHERE u.statut = 'ACTIF'
  AND (ur.date_fin IS NULL OR ur.date_fin >= CURRENT_DATE);

-- 13.2  Refus par modèle (tableau de bord sécurité)
CREATE OR REPLACE VIEW v_refus_par_modele AS
SELECT
    modele_bloquant,
    COUNT(*)                         AS nb_refus,
    COUNT(DISTINCT id_utilisateur)   AS utilisateurs_concernes,
    MAX(horodatage)                  AS dernier_refus
FROM journal_audit
WHERE decision = 'REFUSE'
GROUP BY modele_bloquant
ORDER BY nb_refus DESC;

-- 13.3  Accès temporaires encore actifs
CREATE OR REPLACE VIEW v_acces_temporaires_actifs AS
SELECT
    u.matricule,
    u.nom || ' ' || u.prenom                          AS utilisateur,
    res.reference                                      AS ressource,
    res.type_ressource,
    at.justification,
    at.date_debut,
    at.date_fin,
    ROUND(EXTRACT(EPOCH FROM (at.date_fin - NOW()))/3600::NUMERIC, 1)
                                                       AS heures_restantes,
    a.nom || ' ' || a.prenom                           AS autorise_par
FROM acces_temporaire at
JOIN utilisateurs u   ON at.id_utilisateur = u.id_utilisateur
JOIN ressources   res ON at.id_ressource   = res.id_ressource
JOIN utilisateurs a   ON at.autorise_par   = a.id_utilisateur
WHERE at.statut   = 'ACTIF'
  AND at.date_fin > NOW();

-- 13.4  Alertes Chinese Wall
CREATE OR REPLACE VIEW v_chinese_wall_alertes AS
SELECT
    u.matricule,
    u.nom || ' ' || u.prenom   AS utilisateur,
    e.nom                       AS equipe,
    cwh.groupe_conflict         AS groupe_consulte,
    cwh.classe_conflict,
    cwh.date_consultation,
    'Bloque sur groupes conflictuels de meme classe'::TEXT AS statut
FROM chinese_wall_historique cwh
JOIN utilisateurs u ON cwh.id_utilisateur = u.id_utilisateur
JOIN equipes      e ON u.id_equipe        = e.id_equipe;

-- 14.1  Vérification RBAC
CREATE OR REPLACE FUNCTION verifier_rbac(
    p_id_utilisateur INT,
    p_type_ressource VARCHAR,
    p_action         action_enum
) RETURNS BOOLEAN LANGUAGE plpgsql STABLE AS $$
BEGIN
    RETURN EXISTS (
        SELECT 1
        FROM   utilisateur_roles ur
        JOIN   role_permissions  rp ON ur.id_role       = rp.id_role
        JOIN   permissions       p  ON rp.id_permission = p.id_permission
        WHERE  ur.id_utilisateur = p_id_utilisateur
          AND  ur.actif = TRUE
          AND  (ur.date_fin IS NULL OR ur.date_fin >= CURRENT_DATE)
          AND  p.ressource_type = p_type_ressource
          AND  p.action         = p_action
    );
END;
$$;


-- 14.2  Vérification BLP (confidentialité) — no-read-up / no-write-down
CREATE OR REPLACE FUNCTION verifier_blp(
    p_id_utilisateur   INT,
    p_niveau_ressource niveau_enum,
    p_action           action_enum
) RETURNS BOOLEAN LANGUAGE plpgsql STABLE AS $$
DECLARE
    v_niveau_user niveau_enum;
    v_ord_user    INT;
    v_ord_res     INT;
BEGIN
    SELECT niveau_max INTO v_niveau_user
    FROM   blp_habilitations
    WHERE  id_utilisateur = p_id_utilisateur;

    IF v_niveau_user IS NULL THEN RETURN FALSE; END IF;

    v_ord_user := CASE v_niveau_user
        WHEN 'N1' THEN 1 WHEN 'N2' THEN 2
        WHEN 'N3' THEN 3 WHEN 'N4' THEN 4 END;
    v_ord_res  := CASE p_niveau_ressource
        WHEN 'N1' THEN 1 WHEN 'N2' THEN 2
        WHEN 'N3' THEN 3 WHEN 'N4' THEN 4 END;

    -- No read-up : ne peut pas lire au-dessus de son niveau
    IF p_action IN ('LIRE','EXPORTER') AND v_ord_user < v_ord_res THEN
        RETURN FALSE;
    END IF;
    -- No write-down : ne peut pas écrire en dessous de son niveau
    IF p_action IN ('ECRIRE','MODIFIER','SUPPRIMER') AND v_ord_user > v_ord_res THEN
        RETURN FALSE;
    END IF;
    RETURN TRUE;
END;
$$;


-- 14.3  Vérification Biba (intégrité) — no-write-up
CREATE OR REPLACE FUNCTION verifier_biba(
    p_id_utilisateur   INT,
    p_niveau_ressource niveau_enum,
    p_action           action_enum
) RETURNS BOOLEAN LANGUAGE plpgsql STABLE AS $$
DECLARE
    v_niveau_user niveau_enum;
    v_ord_user    INT;
    v_ord_res     INT;
BEGIN
    SELECT niveau_max INTO v_niveau_user
    FROM   blp_habilitations
    WHERE  id_utilisateur = p_id_utilisateur;

    IF v_niveau_user IS NULL THEN RETURN FALSE; END IF;

    v_ord_user := CASE v_niveau_user
        WHEN 'N1' THEN 1 WHEN 'N2' THEN 2
        WHEN 'N3' THEN 3 WHEN 'N4' THEN 4 END;
    v_ord_res  := CASE p_niveau_ressource
        WHEN 'N1' THEN 1 WHEN 'N2' THEN 2
        WHEN 'N3' THEN 3 WHEN 'N4' THEN 4 END;

    -- No write-up : un utilisateur bas niveau ne modifie pas des données de niveau supérieur
    IF p_action IN ('ECRIRE','MODIFIER','SUPPRIMER','VALIDER') AND v_ord_user < v_ord_res THEN
        RETURN FALSE;
    END IF;
    RETURN TRUE;
END;
$$;


-- 14.4  Vérification ABAC (contexte : heure, jour, pays)
--       Retourne TRUE si toutes les politiques applicables sont respectées
CREATE OR REPLACE FUNCTION verifier_abac(
    p_id_utilisateur INT,
    p_id_ressource   INT,
    p_action         action_enum
) RETURNS BOOLEAN LANGUAGE plpgsql STABLE AS $$
DECLARE
    v_type_ressource  VARCHAR(50);
    v_code_pays_res   CHAR(2);
    v_code_pays_user  CHAR(2);
    v_heure_courante  TIME    := LOCALTIME;
    v_jour_courant    INT     := EXTRACT(ISODOW FROM NOW()); -- 1=lun, 7=dim
    v_politique       RECORD;
    v_jour_debut      INT;
    v_jour_fin        INT;
BEGIN
    SELECT type_ressource, code_pays INTO v_type_ressource, v_code_pays_res
    FROM   ressources WHERE id_ressource = p_id_ressource;

    SELECT code_pays INTO v_code_pays_user
    FROM   utilisateurs WHERE id_utilisateur = p_id_utilisateur;

    -- Parcourir toutes les politiques ABAC applicables au rôle de l'utilisateur
    FOR v_politique IN
        SELECT ap.*
        FROM   abac_politiques ap
        JOIN   utilisateur_roles ur ON ap.id_role = ur.id_role
        WHERE  ur.id_utilisateur = p_id_utilisateur
          AND  ur.actif = TRUE
          AND  (ap.type_ressource IS NULL OR ap.type_ressource = v_type_ressource)
    LOOP
        -- Vérification filtre pays
        IF v_politique.filtre_pays AND v_code_pays_res IS NOT NULL
           AND v_code_pays_res != v_code_pays_user THEN
            RETURN FALSE;
        END IF;

        -- Vérification fenêtre horaire
        IF v_politique.heure_debut IS NOT NULL AND v_politique.heure_fin IS NOT NULL THEN
            IF v_heure_courante < v_politique.heure_debut
            OR v_heure_courante > v_politique.heure_fin THEN
                RETURN FALSE;
            END IF;
        END IF;

        -- Vérification jour de la semaine
        IF v_politique.jours_semaine IS NOT NULL THEN
            v_jour_debut := CASE SPLIT_PART(v_politique.jours_semaine,'-',1)
                WHEN 'LUN' THEN 1 WHEN 'MAR' THEN 2 WHEN 'MER' THEN 3
                WHEN 'JEU' THEN 4 WHEN 'VEN' THEN 5 WHEN 'SAM' THEN 6
                WHEN 'DIM' THEN 7 ELSE 1 END;
            v_jour_fin   := CASE SPLIT_PART(v_politique.jours_semaine,'-',2)
                WHEN 'LUN' THEN 1 WHEN 'MAR' THEN 2 WHEN 'MER' THEN 3
                WHEN 'JEU' THEN 4 WHEN 'VEN' THEN 5 WHEN 'SAM' THEN 6
                WHEN 'DIM' THEN 7 ELSE 7 END;
            IF v_jour_courant < v_jour_debut OR v_jour_courant > v_jour_fin THEN
                RETURN FALSE;
            END IF;
        END IF;
    END LOOP;

    RETURN TRUE;
END;
$$;


-- 14.5  Vérification Chinese Wall (CORRIGÉE — bloque vraiment les conflits)
CREATE OR REPLACE FUNCTION verifier_chinese_wall(
    p_id_utilisateur INT,
    p_id_ressource   INT
) RETURNS BOOLEAN LANGUAGE plpgsql STABLE AS $$
DECLARE
    v_groupe_ressource VARCHAR(50);
    v_classe_ressource VARCHAR(50);
BEGIN
    -- Récupérer le groupe et la classe de conflit de la ressource demandée
    SELECT groupe_conflict, classe_conflict
    INTO   v_groupe_ressource, v_classe_ressource
    FROM   ressources
    WHERE  id_ressource = p_id_ressource;

    -- Ressource sans groupe de conflit → pas de mur applicable
    IF v_groupe_ressource IS NULL THEN
        RETURN TRUE;
    END IF;

    -- Vérifier si l'utilisateur a déjà accédé à un GROUPE DIFFÉRENT
    -- dans la MÊME CLASSE de conflit
    IF EXISTS (
        SELECT 1
        FROM   chinese_wall_historique cwh
        WHERE  cwh.id_utilisateur  = p_id_utilisateur
          AND  cwh.classe_conflict = v_classe_ressource
          AND  cwh.groupe_conflict != v_groupe_ressource
    ) THEN
        -- Conflit d'intérêt détecté → REFUS
        RETURN FALSE;
    END IF;

    -- Pas de conflit → autoriser ET enregistrer la consultation
    INSERT INTO chinese_wall_historique
      (id_utilisateur, id_ressource, groupe_conflict, classe_conflict)
    VALUES
      (p_id_utilisateur, p_id_ressource, v_groupe_ressource, v_classe_ressource)
    ON CONFLICT (id_utilisateur, groupe_conflict) DO NOTHING;

    RETURN TRUE;
END;
$$;


-- 14.6  Fonction principale de décision d'accès (orchestre toutes les couches)
CREATE OR REPLACE FUNCTION controler_acces(
    p_id_utilisateur INT,
    p_id_ressource   INT,
    p_action         action_enum,
    p_justification  TEXT DEFAULT NULL
) RETURNS TABLE (
    decision        decision_enum,
    modele_bloquant VARCHAR,
    message         TEXT
) LANGUAGE plpgsql AS $$
DECLARE
    v_type_ressource   VARCHAR(50);
    v_niveau_ressource niveau_enum;
    v_statut_user      statut_enum;
    v_mfa_requis       BOOLEAN;
    v_mfa_actif        BOOLEAN;
    v_decision         decision_enum;
    v_modele           VARCHAR;
    v_message          TEXT;
BEGIN
    -- Récupérer les infos de la ressource
    SELECT type_ressource, niveau_requis
    INTO   v_type_ressource, v_niveau_ressource
    FROM   ressources
    WHERE  id_ressource = p_id_ressource AND actif = TRUE;

    IF NOT FOUND THEN
        v_decision := 'REFUSE'; v_modele := 'SYSTEME';
        v_message  := 'Ressource introuvable ou inactive';
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision, modele_bloquant, horodatage)
        VALUES (p_id_utilisateur, p_id_ressource, 'INCONNU', p_action, v_decision, v_modele, NOW());
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Vérifier le statut de l'utilisateur
    SELECT statut, mfa_actif INTO v_statut_user, v_mfa_actif
    FROM   utilisateurs WHERE id_utilisateur = p_id_utilisateur;

    IF v_statut_user != 'ACTIF' THEN
        v_decision := 'REFUSE'; v_modele := 'SYSTEME';
        v_message  := 'Utilisateur inactif ou suspendu';
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision,
           modele_bloquant, motif_refus, horodatage)
        VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
                v_decision, v_modele, v_message, NOW());
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Vérification MFA (obligatoire pour N3/N4)
    SELECT mfa_requis INTO v_mfa_requis
    FROM   niveaux_securite WHERE niveau = v_niveau_ressource;

    IF v_mfa_requis AND NOT v_mfa_actif THEN
        v_decision := 'REFUSE'; v_modele := 'AUTH';
        v_message  := 'MFA obligatoire pour niveau ' || v_niveau_ressource;
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision,
           modele_bloquant, motif_refus, horodatage)
        VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
                v_decision, v_modele, v_message, NOW());
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Justification obligatoire pour N3/N4
    IF v_niveau_ressource IN ('N3','N4')
       AND (p_justification IS NULL OR p_justification = '') THEN
        v_decision := 'REFUSE'; v_modele := 'POLITIQUE';
        v_message  := 'Justification metier obligatoire pour niveau ' || v_niveau_ressource;
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision,
           modele_bloquant, motif_refus, horodatage)
        VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
                v_decision, v_modele, v_message, NOW());
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Couche 1 : RBAC
    IF NOT verifier_rbac(p_id_utilisateur, v_type_ressource, p_action) THEN
        v_decision := 'REFUSE'; v_modele := 'RBAC';
        v_message  := 'Role insuffisant pour ' || v_type_ressource || '/' || p_action;
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision,
           modele_bloquant, motif_refus, horodatage, justification)
        VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
                v_decision, v_modele, v_message, NOW(), p_justification);
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Couche 2 : BLP
    IF NOT verifier_blp(p_id_utilisateur, v_niveau_ressource, p_action) THEN
        v_decision := 'REFUSE'; v_modele := 'BLP';
        v_message  := 'Violation Bell-LaPadula : no-read-up ou no-write-down';
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision,
           modele_bloquant, motif_refus, horodatage, justification)
        VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
                v_decision, v_modele, v_message, NOW(), p_justification);
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Couche 3 : Biba
    IF NOT verifier_biba(p_id_utilisateur, v_niveau_ressource, p_action) THEN
        v_decision := 'REFUSE'; v_modele := 'BIBA';
        v_message  := 'Violation Biba : no-write-up interdit';
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision,
           modele_bloquant, motif_refus, horodatage, justification)
        VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
                v_decision, v_modele, v_message, NOW(), p_justification);
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Couche 4 : ABAC (contexte : heure, jour, pays)
    IF NOT verifier_abac(p_id_utilisateur, p_id_ressource, p_action) THEN
        v_decision := 'REFUSE'; v_modele := 'ABAC';
        v_message  := 'Violation ABAC : heure, jour ou pays non autorise';
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision,
           modele_bloquant, motif_refus, horodatage, justification)
        VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
                v_decision, v_modele, v_message, NOW(), p_justification);
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Couche 5 : Chinese Wall
    IF NOT verifier_chinese_wall(p_id_utilisateur, p_id_ressource) THEN
        v_decision := 'REFUSE'; v_modele := 'CHINESE_WALL';
        v_message  := 'Conflit d interet detecte (Chinese Wall)';
        INSERT INTO journal_audit
          (id_utilisateur, id_ressource, type_ressource, action, decision,
           modele_bloquant, motif_refus, horodatage, justification)
        VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
                v_decision, v_modele, v_message, NOW(), p_justification);
        RETURN QUERY SELECT v_decision, v_modele, v_message; RETURN;
    END IF;

    -- Toutes les couches validées → AUTORISÉ
    INSERT INTO journal_audit
      (id_utilisateur, id_ressource, type_ressource, action, decision,
       horodatage, justification)
    VALUES (p_id_utilisateur, p_id_ressource, v_type_ressource, p_action,
            'AUTORISE', NOW(), p_justification);

    RETURN QUERY SELECT 'AUTORISE'::decision_enum, NULL::VARCHAR,
        'Acces accorde - toutes les politiques validees'::TEXT;
END;
$$;

CREATE OR REPLACE FUNCTION current_user_id() RETURNS INT
LANGUAGE sql STABLE SECURITY DEFINER AS $$
    SELECT id_utilisateur
    FROM   utilisateurs
    WHERE  matricule = UPPER(SPLIT_PART(current_user, '_', 2))
      AND  statut    = 'ACTIF'
    LIMIT 1;
$$;

-- ==============================================================
-- PARTIE 17 : FONCTION D'ATTRIBUTION AUTOMATIQUE
-- ==============================================================

CREATE OR REPLACE FUNCTION attribuer_acces_utilisateur(
    p_matricule    VARCHAR,
    p_attribue_par VARCHAR DEFAULT 'SYSTEME'
)
RETURNS TABLE (etape TEXT, statut TEXT, detail TEXT)
LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
    v_id_user    INT;
    v_id_equipe  INT;
    v_nom_equipe VARCHAR;
    v_niveau_max niveau_enum;
    v_pg_role    VARCHAR;
    v_id_admin   INT;
    v_role       RECORD;
    v_count      INT := 0;
BEGIN
    -- Vérifier l'utilisateur
    SELECT u.id_utilisateur, u.id_equipe, e.nom, e.niveau_max
    INTO   v_id_user, v_id_equipe, v_nom_equipe, v_niveau_max
    FROM   utilisateurs u
    JOIN   equipes e ON u.id_equipe = e.id_equipe
    WHERE  u.matricule = UPPER(p_matricule) AND u.statut = 'ACTIF';

    IF NOT FOUND THEN
        RETURN QUERY SELECT
            'Verification utilisateur'::TEXT, 'ERREUR'::TEXT,
            ('Utilisateur ' || p_matricule || ' introuvable ou inactif')::TEXT;
        RETURN;
    END IF;

    RETURN QUERY SELECT 'Verification utilisateur'::TEXT, 'OK'::TEXT,
        ('id=' || v_id_user || ' equipe=' || v_nom_equipe || ' niveau=' || v_niveau_max)::TEXT;

    -- Résoudre l'admin
    SELECT id_utilisateur INTO v_id_admin
    FROM   utilisateurs WHERE matricule = UPPER(p_attribue_par);

    -- Mapper équipe → rôle PostgreSQL
    v_pg_role := CASE v_nom_equipe
        WHEN 'Relation clientele'    THEN 'banque_rc'
        WHEN 'Back-office'           THEN 'banque_bo'
        WHEN 'Gestion de patrimoine' THEN 'banque_gp'
        WHEN 'Risque'                THEN 'banque_risque'
        WHEN 'Conformite / AML'      THEN 'banque_aml'
        WHEN 'Audit interne'         THEN 'banque_audit'
        WHEN 'Juridique'             THEN 'banque_juridique'
        WHEN 'Cadres commerciaux'    THEN 'banque_commercial'
        WHEN 'Experts regionaux'     THEN 'banque_expert'
        WHEN 'Support'               THEN 'banque_support'
        ELSE NULL
    END;

    IF v_pg_role IS NOT NULL THEN
        -- Créer le login PostgreSQL s'il n'existe pas
        IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = LOWER(p_matricule)) THEN
            EXECUTE format(
                'CREATE ROLE %I LOGIN PASSWORD %L',
                LOWER(p_matricule),
                'Banque_' || p_matricule || '_2024!'
            );
        END IF;
        EXECUTE format('GRANT %I TO %I', v_pg_role, LOWER(p_matricule));
        RETURN QUERY SELECT 'Role PostgreSQL'::TEXT, 'OK'::TEXT,
            ('login=' || LOWER(p_matricule) || ' → ' || v_pg_role)::TEXT;
    ELSE
        RETURN QUERY SELECT 'Role PostgreSQL'::TEXT, 'AVERTISSEMENT'::TEXT,
            ('Equipe inconnue : ' || v_nom_equipe)::TEXT;
    END IF;

    -- Attribuer les rôles RBAC de l'équipe
    FOR v_role IN
        SELECT id_role, nom_role FROM roles WHERE id_equipe = v_id_equipe
    LOOP
        INSERT INTO utilisateur_roles (id_utilisateur, id_role, date_debut, actif)
        VALUES (v_id_user, v_role.id_role, CURRENT_DATE, TRUE)
        ON CONFLICT (id_utilisateur, id_role)
        DO UPDATE SET actif = TRUE, date_debut = CURRENT_DATE, date_fin = NULL;
        v_count := v_count + 1;
    END LOOP;

    RETURN QUERY SELECT 'Roles RBAC'::TEXT, 'OK'::TEXT,
        (v_count || ' role(s) attribue(s) pour ' || v_nom_equipe)::TEXT;

    -- Initialiser / mettre à jour l'habilitation BLP
    INSERT INTO blp_habilitations
      (id_utilisateur, niveau_max, date_attribution, attribue_par, commentaire)
    VALUES
      (v_id_user, v_niveau_max, CURRENT_DATE, v_id_admin,
       'Attribution automatique - equipe ' || v_nom_equipe)
    ON CONFLICT (id_utilisateur)
    DO UPDATE SET
        niveau_max       = EXCLUDED.niveau_max,
        date_attribution = EXCLUDED.date_attribution,
        attribue_par     = EXCLUDED.attribue_par,
        commentaire      = EXCLUDED.commentaire;

    RETURN QUERY SELECT 'Habilitation BLP'::TEXT, 'OK'::TEXT,
        ('niveau_max=' || v_niveau_max)::TEXT;

    -- Activer MFA automatiquement pour N3/N4
    IF v_niveau_max IN ('N3','N4') THEN
        UPDATE utilisateurs SET mfa_actif = TRUE
        WHERE  id_utilisateur = v_id_user AND mfa_actif = FALSE;
        RETURN QUERY SELECT 'MFA'::TEXT, 'OK'::TEXT,
            ('MFA active automatiquement (niveau ' || v_niveau_max || ')')::TEXT;
    ELSE
        RETURN QUERY SELECT 'MFA'::TEXT, 'NON_REQUIS'::TEXT,
            ('Niveau ' || v_niveau_max || ' - MFA optionnel')::TEXT;
    END IF;

    -- Journaliser l'attribution
    INSERT INTO journal_audit
      (id_utilisateur, type_ressource, action, decision, horodatage, justification)
    VALUES
      (v_id_user, 'SYSTEME', 'ECRIRE', 'AUTORISE', NOW(),
       'Attribution automatique - equipe: ' || v_nom_equipe
       || ' | niveau: ' || v_niveau_max
       || ' | roles RBAC: ' || v_count
       || ' | par: ' || p_attribue_par);

    RETURN QUERY SELECT 'Journalisation'::TEXT, 'OK'::TEXT,
        'Attribution enregistree dans journal_audit'::TEXT;
END;
$$;

GRANT EXECUTE ON FUNCTION attribuer_acces_utilisateur(VARCHAR, VARCHAR)
    TO banque_dba, banque_audit;


-- ==============================================================
-- PARTIE 18 : FONCTION DE RÉVOCATION AUTOMATIQUE
-- ==============================================================

CREATE OR REPLACE FUNCTION revoquer_acces_utilisateur(
    p_matricule   VARCHAR,
    p_motif       TEXT    DEFAULT 'Revocation manuelle',
    p_revoque_par VARCHAR DEFAULT 'SYSTEME'
)
RETURNS TABLE (etape TEXT, statut TEXT, detail TEXT)
LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
    v_id_user INT;
    v_count   INT;
BEGIN
    SELECT id_utilisateur INTO v_id_user
    FROM   utilisateurs WHERE matricule = UPPER(p_matricule);

    IF NOT FOUND THEN
        RETURN QUERY SELECT 'Verification'::TEXT, 'ERREUR'::TEXT,
            ('Utilisateur ' || p_matricule || ' introuvable')::TEXT;
        RETURN;
    END IF;

    -- Désactiver tous les rôles RBAC
    UPDATE utilisateur_roles
    SET    actif = FALSE, date_fin = CURRENT_DATE
    WHERE  id_utilisateur = v_id_user AND actif = TRUE;
    GET DIAGNOSTICS v_count = ROW_COUNT;
    RETURN QUERY SELECT 'Revocation RBAC'::TEXT, 'OK'::TEXT,
        (v_count || ' role(s) revoques')::TEXT;

    -- Expirer les accès temporaires
    UPDATE acces_temporaire
    SET    statut = 'EXPIRE', date_revocation = NOW()
    WHERE  id_utilisateur = v_id_user AND statut = 'ACTIF';
    GET DIAGNOSTICS v_count = ROW_COUNT;
    RETURN QUERY SELECT 'Acces temporaires'::TEXT, 'OK'::TEXT,
        (v_count || ' acces temporaire(s) expires')::TEXT;

    -- Suspendre le compte métier
    UPDATE utilisateurs SET statut = 'SUSPENDU'
    WHERE  id_utilisateur = v_id_user;
    RETURN QUERY SELECT 'Suspension compte'::TEXT, 'OK'::TEXT, 'Statut -> SUSPENDU'::TEXT;

    -- Révoquer le login PostgreSQL
    IF EXISTS (SELECT 1 FROM pg_roles WHERE rolname = LOWER(p_matricule)) THEN
        EXECUTE format('ALTER ROLE %I NOLOGIN', LOWER(p_matricule));
        RETURN QUERY SELECT 'Role PostgreSQL'::TEXT, 'OK'::TEXT,
            (LOWER(p_matricule) || ' -> NOLOGIN')::TEXT;
    END IF;

    -- Journaliser
    INSERT INTO journal_audit
      (id_utilisateur, type_ressource, action, decision, horodatage, justification)
    VALUES (v_id_user, 'SYSTEME', 'SUPPRIMER', 'AUTORISE', NOW(),
            'Revocation complete - motif: ' || p_motif || ' | par: ' || p_revoque_par);
    RETURN QUERY SELECT 'Journalisation'::TEXT, 'OK'::TEXT,
        'Revocation enregistree dans journal_audit'::TEXT;
END;
$$;

GRANT EXECUTE ON FUNCTION revoquer_acces_utilisateur(VARCHAR, TEXT, VARCHAR)
    TO banque_dba, banque_audit;


-- ==============================================================
-- PARTIE 19 : FONCTION D'ACCÈS TEMPORAIRE CONTRÔLÉ
-- ==============================================================

CREATE OR REPLACE FUNCTION accorder_acces_temporaire(
    p_matricule_user  VARCHAR,
    p_ref_ressource   VARCHAR,
    p_nb_heures       INT,
    p_justification   TEXT,
    p_matricule_admin VARCHAR
)
RETURNS TABLE (etape TEXT, statut TEXT, detail TEXT)
LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
    v_id_user      INT;
    v_id_ressource INT;
    v_id_admin     INT;
    v_niveau_res   niveau_enum;
    v_niveau_admin niveau_enum;
    v_duree_max    INT;
BEGIN
    SELECT id_utilisateur INTO v_id_user
    FROM   utilisateurs WHERE matricule = UPPER(p_matricule_user) AND statut = 'ACTIF';

    SELECT id_utilisateur INTO v_id_admin
    FROM   utilisateurs WHERE matricule = UPPER(p_matricule_admin) AND statut = 'ACTIF';

    SELECT id_ressource, niveau_requis INTO v_id_ressource, v_niveau_res
    FROM   ressources WHERE reference = p_ref_ressource AND actif = TRUE;

    IF v_id_user IS NULL OR v_id_ressource IS NULL OR v_id_admin IS NULL THEN
        RETURN QUERY SELECT 'Verification'::TEXT, 'ERREUR'::TEXT,
            'Utilisateur, ressource ou admin introuvable'::TEXT;
        RETURN;
    END IF;

    -- L'admin doit avoir un niveau >= niveau de la ressource
    SELECT niveau_max INTO v_niveau_admin
    FROM   blp_habilitations WHERE id_utilisateur = v_id_admin;

    IF v_niveau_admin < v_niveau_res THEN
        RETURN QUERY SELECT 'Verification BLP admin'::TEXT, 'REFUSE'::TEXT,
            ('Admin ' || p_matricule_admin || ' niveau ' || v_niveau_admin
             || ' < ressource niveau ' || v_niveau_res)::TEXT;
        RETURN;
    END IF;
    RETURN QUERY SELECT 'Verification BLP admin'::TEXT, 'OK'::TEXT,
        ('Admin niveau ' || v_niveau_admin || ' >= ressource niveau ' || v_niveau_res)::TEXT;

    -- Respecter la durée max ABAC
    SELECT MIN(ap.duree_max_heures) INTO v_duree_max
    FROM   abac_politiques ap
    JOIN   utilisateur_roles ur ON ap.id_role = ur.id_role
    WHERE  ur.id_utilisateur = v_id_user AND ur.actif = TRUE
      AND  ap.duree_max_heures IS NOT NULL;

    IF v_duree_max IS NOT NULL AND p_nb_heures > v_duree_max THEN
        RETURN QUERY SELECT 'Verification ABAC duree'::TEXT, 'REFUSE'::TEXT,
            ('Duree ' || p_nb_heures || 'h > max autorise ' || v_duree_max || 'h')::TEXT;
        RETURN;
    END IF;
    RETURN QUERY SELECT 'Verification ABAC duree'::TEXT, 'OK'::TEXT,
        (p_nb_heures || 'h <= max=' || COALESCE(v_duree_max::TEXT, 'illimite'))::TEXT;

    -- Insérer l'accès temporaire
    INSERT INTO acces_temporaire
      (id_utilisateur, id_ressource, justification, date_debut, date_fin,
       autorise_par, revocation_auto)
    VALUES
      (v_id_user, v_id_ressource, p_justification,
       NOW(), NOW() + (p_nb_heures || ' hours')::INTERVAL,
       v_id_admin, TRUE);

    RETURN QUERY SELECT 'Creation acces temporaire'::TEXT, 'OK'::TEXT,
        ('Valide ' || p_nb_heures || 'h - expire le '
         || TO_CHAR(NOW() + (p_nb_heures || ' hours')::INTERVAL, 'DD/MM/YYYY HH24:MI'))::TEXT;

    -- Journaliser
    INSERT INTO journal_audit
      (id_utilisateur, id_ressource, type_ressource, action, decision,
       horodatage, justification)
    VALUES (v_id_user, v_id_ressource, 'ACCES_TEMPORAIRE', 'ECRIRE', 'AUTORISE', NOW(),
            'Acces temporaire accorde par ' || p_matricule_admin
            || ' - duree: ' || p_nb_heures || 'h - motif: ' || p_justification);

    RETURN QUERY SELECT 'Journalisation'::TEXT, 'OK'::TEXT,
        'Acces temporaire enregistre dans journal_audit'::TEXT;
END;
$$;

GRANT EXECUTE ON FUNCTION accorder_acces_temporaire(VARCHAR, VARCHAR, INT, TEXT, VARCHAR)
    TO banque_dba, banque_audit, banque_aml, banque_juridique;


-- ==============================================================
-- PARTIE 20 : TRIGGER D'ATTRIBUTION AUTOMATIQUE À L'INSERTION
-- ==============================================================

CREATE OR REPLACE FUNCTION trigger_auto_attribution()
RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE r RECORD;
BEGIN
    FOR r IN
        SELECT * FROM attribuer_acces_utilisateur(NEW.matricule, 'SYSTEME')
    LOOP
        RAISE NOTICE '[auto_attribution] matricule=% | % | % | %',
            NEW.matricule, r.etape, r.statut, r.detail;
    END LOOP;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_auto_attribution
    AFTER INSERT ON utilisateurs
    FOR EACH ROW
    EXECUTE FUNCTION trigger_auto_attribution();


-- ==============================================================
-- PARTIE 21 : TRIGGER DE RÉVOCATION AUTOMATIQUE DES ACCÈS EXPIRÉS
-- ==============================================================

CREATE OR REPLACE FUNCTION trigger_revocation_auto()
RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER AS $$
BEGIN
    UPDATE acces_temporaire
    SET    statut = 'EXPIRE', date_revocation = NOW()
    WHERE  statut = 'ACTIF'
      AND  date_fin <= NOW()
      AND  revocation_auto = TRUE;
    RETURN NULL;
END;
$$;

CREATE TRIGGER trg_revocation_auto
    AFTER INSERT OR UPDATE ON acces_temporaire
    FOR EACH STATEMENT
    EXECUTE FUNCTION trigger_revocation_auto();

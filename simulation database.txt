-- -------------------------------------------------------------
-- 1. TYPES ENUMERES
-- -------------------------------------------------------------

CREATE TYPE niveau_enum   AS ENUM ('N1','N2','N3','N4');
CREATE TYPE action_enum   AS ENUM ('LIRE','ECRIRE','MODIFIER','SUPPRIMER','VALIDER','EXPORTER');
CREATE TYPE decision_enum AS ENUM ('AUTORISE','REFUSE','EN_ATTENTE');
CREATE TYPE statut_enum   AS ENUM ('ACTIF','INACTIF','SUSPENDU','EXPIRE');

-- 3.3  Association roles <-> permissions
CREATE TABLE role_permissions (
    id_role         INT     NOT NULL REFERENCES roles(id_role)       ON DELETE CASCADE,
    id_permission   INT     NOT NULL REFERENCES permissions(id_permission) ON DELETE CASCADE,
    PRIMARY KEY (id_role, id_permission)
);

-- Relation clientele
INSERT INTO role_permissions
  SELECT r.id_role, p.id_permission
  FROM roles r, permissions p
  WHERE r.nom_role IN ('conseiller_particulier','conseiller_entreprise')
    AND p.ressource_type IN ('DONNEES_CLIENT','DOSSIER_KYC','OPERATIONS_COURANTES','CONTRAT_CREDIT')
    AND p.action IN ('LIRE','ECRIRE');

INSERT INTO role_permissions
  SELECT r.id_role, p.id_permission
  FROM roles r, permissions p
  WHERE r.nom_role = 'responsable_agence'

-- Vue : droits effectifs d'un utilisateur (RBAC)
CREATE OR REPLACE VIEW v_droits_utilisateur AS
SELECT
    u.matricule,
    u.nom || ' ' || u.prenom   AS utilisateur,
    e.nom                       AS equipe,
    r.nom_role                  AS role,
    p.ressource_type,
    p.action,
    p.niveau_min                AS niveau_requis,
    h.niveau_max                AS niveau_habilite
FROM utilisateurs u
JOIN utilisateur_roles ur   ON u.id_utilisateur = ur.id_utilisateur AND ur.actif = TRUE
JOIN roles r                ON ur.id_role = r.id_role
JOIN equipes e              ON r.id_equipe = e.id_equipe
JOIN role_permissions rp    ON r.id_role = rp.id_role
JOIN permissions p          ON rp.id_permission = p.id_permission
JOIN blp_habilitations h    ON u.id_utilisateur = h.id_utilisateur
WHERE u.statut = 'ACTIF'
  AND (ur.date_fin IS NULL OR ur.date_fin >= CURRENT_DATE);


-- Vue : acces refus par modele (tableau de bord securite)
CREATE OR REPLACE VIEW v_refus_par_modele AS
SELECT
    modele_bloquant,
    COUNT(*)                    AS nb_refus,
    COUNT(DISTINCT id_utilisateur) AS utilisateurs_concernes,
    MAX(horodatage)             AS dernier_refus
FROM journal_audit
WHERE decision = 'REFUSE'
GROUP BY modele_bloquant
ORDER BY nb_refus DESC;


-- Vue : acces temporaires actifs
CREATE OR REPLACE VIEW v_acces_temporaires_actifs AS
SELECT
    u.matricule,
    u.nom || ' ' || u.prenom   AS utilisateur,
    res.reference               AS ressource,
    res.type_ressource,
    at.justification,
    at.date_debut,
    at.date_fin,
    EXTRACT(EPOCH FROM (at.date_fin - NOW()))/3600 AS heures_restantes,
    a.nom || ' ' || a.prenom   AS autorise_par
FROM acces_temporaire at
JOIN utilisateurs u     ON at.id_utilisateur = u.id_utilisateur
JOIN ressources res     ON at.id_ressource = res.id_ressource
JOIN utilisateurs a     ON at.autorise_par = a.id_utilisateur
WHERE at.statut = 'ACTIF'
  AND at.date_fin > NOW();


-- Vue : alertes Chinese Wall actives
CREATE OR REPLACE VIEW v_chinese_wall_alertes AS
SELECT
    u.matricule,
    u.nom || ' ' || u.prenom   AS utilisateur,
    e.nom                       AS equipe,
    cwh.groupe_conflict         AS groupe_consulte,

FROM chinese_wall_historique cwh
JOIN utilisateurs u ON cwh.id_utilisateur = u.id_utilisateur
JOIN equipes e      ON u.id_equipe = e.id_equipe;

-- 12.1  Verification RBAC
CREATE OR REPLACE FUNCTION verifier_rbac(
    p_id_utilisateur INT,
    p_type_ressource VARCHAR,
    p_action         action_enum
) RETURNS BOOLEAN AS $$
DECLARE
    v_autorise BOOLEAN := FALSE;
BEGIN
    SELECT EXISTS (
        SELECT 1
        FROM utilisateur_roles ur
        JOIN role_permissions rp ON ur.id_role = rp.id_role
        JOIN permissions p       ON rp.id_permission = p.id_permission
        WHERE ur.id_utilisateur = p_id_utilisateur
          AND ur.actif = TRUE
          AND (ur.date_fin IS NULL OR ur.date_fin >= CURRENT_DATE)
          AND p.ressource_type = p_type_ressource
          AND p.action = p_action
    ) INTO v_autorise;
    RETURN v_autorise;
END;
$$ LANGUAGE plpgsql;


-- 12.2  Verification BLP (confidentialite)
CREATE OR REPLACE FUNCTION verifier_blp(
    p_id_utilisateur INT,
    p_niveau_ressource niveau_enum,
    p_action           action_enum
) RETURNS BOOLEAN AS $$
DECLARE
    v_niveau_user niveau_enum;
    v_ordre_user  INT;
    v_ordre_res   INT;
BEGIN
    SELECT niveau_max INTO v_niveau_user
    FROM blp_habilitations
    WHERE id_utilisateur = p_id_utilisateur;

    IF v_niveau_user IS NULL THEN RETURN FALSE; END IF;

    -- Ordre numerique des niveaux
    v_ordre_user := CASE v_niveau_user
        WHEN 'N1' THEN 1 WHEN 'N2' THEN 2
        WHEN 'N3' THEN 3 WHEN 'N4' THEN 4
    END;
    v_ordre_res := CASE p_niveau_ressource
        WHEN 'N1' THEN 1 WHEN 'N2' THEN 2
        WHEN 'N3' THEN 3 WHEN 'N4' THEN 4
    END;

    -- No read up : l'utilisateur ne peut pas lire au-dessus de son niveau
    IF p_action IN ('LIRE','EXPORTER') AND v_ordre_user < v_ordre_res THEN
        RETURN FALSE;
    END IF;
    -- No write down : l'utilisateur ne peut pas ecrire en dessous de son niveau
    IF p_action IN ('ECRIRE','MODIFIER','SUPPRIMER') AND v_ordre_user > v_ordre_res THEN
        RETURN FALSE;
    END IF;
    RETURN TRUE;
END;
$$ LANGUAGE plpgsql;


-- 12.3  Verification Biba (integrite)
CREATE OR REPLACE FUNCTION verifier_biba(
    p_id_utilisateur   INT,
    p_niveau_ressource niveau_enum,
    p_action           action_enum
) RETURNS BOOLEAN AS $$
DECLARE
    v_niveau_user niveau_enum;
    v_ordre_user  INT;
    v_ordre_res   INT;
BEGIN
    SELECT niveau_max INTO v_niveau_user
    FROM blp_habilitations
    WHERE id_utilisateur = p_id_utilisateur;

    IF v_niveau_user IS NULL THEN RETURN FALSE; END IF;

    v_ordre_user := CASE v_niveau_user
        WHEN 'N1' THEN 1 WHEN 'N2' THEN 2
        WHEN 'N3' THEN 3 WHEN 'N4' THEN 4
    END;
    v_ordre_res := CASE p_niveau_ressource
        WHEN 'N1' THEN 1 WHEN 'N2' THEN 2
        WHEN 'N3' THEN 3 WHEN 'N4' THEN 4
    END;

    -- No write up : un utilisateur ne peut pas ecrire au-dessus de son niveau
    IF p_action IN ('ECRIRE','MODIFIER','SUPPRIMER','VALIDER') AND v_ordre_user < v_ordre_res THEN
        RETURN FALSE;
    END IF;
    -- No read down : un utilisateur de haut niveau ne lit pas le bas (optionnel strict)
    -- Desactive ici pour compatibilite operationnelle (audit doit tout lire)
    RETURN TRUE;
END;
$$ LANGUAGE plpgsql;


-- 12.4  Verification Chinese Wall
CREATE OR REPLACE FUNCTION verifier_chinese_wall(
    p_id_utilisateur INT,
    p_id_ressource   INT
) RETURNS BOOLEAN AS $$
DECLARE
    v_groupe_ressource VARCHAR(50);
BEGIN
    -- Recuperer le groupe de conflit de la ressource demandee
    -- (En production, la table ressources aurait une colonne groupe_conflict)
    -- Ici on verifie si l'utilisateur a deja consulte un groupe conflictuel
    SELECT groupe_conflict INTO v_groupe_ressource
    FROM chinese_wall_historique
    WHERE id_utilisateur = p_id_utilisateur
      AND id_ressource = p_id_ressource
    LIMIT 1;

    -- Si aucun conflit enregistre, acces autorise
    IF v_groupe_ressource IS NULL THEN
        RETURN TRUE;
    END IF;

    -- Si le meme groupe : autoriser (on l'a deja consulte)
    -- Si groupe different dans meme classe de conflit : refuser
    -- Logique simplifiee : si l'utilisateur est dans le registre pour ce groupe, OK
    RETURN TRUE;
END;
$$ LANGUAGE plpgsql;


-- 12.5  Fonction principale de decision d'acces (orchestre tous les modeles)
CREATE OR REPLACE FUNCTION controler_acces(
    p_id_utilisateur INT,
    p_id_ressource   INT,
    p_action         action_enum,
    p_justification  TEXT DEFAULT NULL
) RETURNS TABLE (
    decision        decision_enum,
    modele_bloquant VARCHAR,
    message         TEXT
) AS $$
DECLARE
    v_type_ressource VARCHAR(50);
    v_niveau_ressource niveau_enum;
    v_statut_user    statut_enum;
    v_mfa_requis     BOOLEAN;
    v_mfa_actif      BOOLEAN;
BEGIN
    -- Recuperer infos ressource
    SELECT type_ressource, niveau_requis
    INTO v_type_ressource, v_niveau_ressource
    FROM ressources WHERE id_ressource = p_id_ressource AND actif = TRUE;

    IF NOT FOUND THEN
        RETURN QUERY SELECT 'REFUSE'::decision_enum, 'SYSTEME'::VARCHAR,
            'Ressource introuvable ou inactive'::TEXT;
        RETURN;
    END IF;

    -- Verifier statut utilisateur
    SELECT statut, mfa_actif INTO v_statut_user, v_mfa_actif
    FROM utilisateurs WHERE id_utilisateur = p_id_utilisateur;

    IF v_statut_user != 'ACTIF' THEN
        RETURN QUERY SELECT 'REFUSE'::decision_enum, 'SYSTEME'::VARCHAR,
            'Utilisateur inactif ou suspendu'::TEXT;
        RETURN;
    END IF;

    -- Verifier MFA si niveau N3/N4
    SELECT mfa_requis INTO v_mfa_requis FROM niveaux_securite WHERE niveau = v_niveau_ressource;
    IF v_mfa_requis AND NOT v_mfa_actif THEN
        RETURN QUERY SELECT 'REFUSE'::decision_enum, 'AUTH'::VARCHAR,
            'MFA obligatoire pour ce niveau de securite'::TEXT;
        RETURN;
    END IF;

    -- Justification obligatoire pour N3/N4
    IF v_niveau_ressource IN ('N3','N4') AND (p_justification IS NULL OR p_justification = '') THEN
        RETURN QUERY SELECT 'REFUSE'::decision_enum, 'POLITIQUE'::VARCHAR,
            'Justification metier obligatoire pour niveau N3/N4'::TEXT;
        RETURN;
    END IF;

    -- Couche 1 : RBAC
    IF NOT verifier_rbac(p_id_utilisateur, v_type_ressource, p_action) THEN
        RETURN QUERY SELECT 'REFUSE'::decision_enum, 'RBAC'::VARCHAR,
            'Role insuffisant pour cette ressource/action'::TEXT;
        RETURN;
    END IF;

    -- Couche 2 : BLP
    IF NOT verifier_blp(p_id_utilisateur, v_niveau_ressource, p_action) THEN
        RETURN QUERY SELECT 'REFUSE'::decision_enum, 'BLP'::VARCHAR,
            'Violation Bell-LaPadula : no-read-up ou no-write-down'::TEXT;
        RETURN;
    END IF;

    -- Couche 3 : Biba
    IF NOT verifier_biba(p_id_utilisateur, v_niveau_ressource, p_action) THEN
        RETURN QUERY SELECT 'REFUSE'::decision_enum, 'BIBA'::VARCHAR,
            'Violation Biba : no-write-up interdit'::TEXT;
        RETURN;
    END IF;

    -- Couche 4 : Chinese Wall
    IF NOT verifier_chinese_wall(p_id_utilisateur, p_id_ressource) THEN
        RETURN QUERY SELECT 'REFUSE'::decision_enum, 'CHINESE_WALL'::VARCHAR,
            'Conflit d interet detecte (Chinese Wall)'::TEXT;
        RETURN;
    END IF;

    -- Toutes les couches validees
    RETURN QUERY SELECT 'AUTORISE'::decision_enum, NULL::VARCHAR,
        'Acces accorde - toutes les politiques validees'::TEXT;
END;
$$ LANGUAGE plpgsql;

-- 13. TRIGGER : JOURNALISATION AUTOMATIQUE
-- -------------------------------------------------------------

CREATE OR REPLACE FUNCTION trigger_journal_acces()
RETURNS TRIGGER AS $$
BEGIN
    -- Ce trigger s'activerait sur une table "demandes_acces" en production
    -- Ici on montre le pattern
    INSERT INTO journal_audit
        (id_utilisateur, id_ressource, type_ressource, action, decision, horodatage)
    VALUES
        (NEW.id_utilisateur, NEW.id_ressource, 'SYSTEME', 'LIRE', 'AUTORISE', NOW());
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


-- -------------------------------------------------------------
-- 14. TRIGGER : REVOCATION AUTOMATIQUE ACCES TEMPORAIRES
-- -------------------------------------------------------------

CREATE OR REPLACE FUNCTION trigger_revocation_auto()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE acces_temporaire
    SET statut = 'EXPIRE', date_revocation = NOW()
    WHERE statut = 'ACTIF'
      AND date_fin <= NOW()
      AND revocation_auto = TRUE;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;
